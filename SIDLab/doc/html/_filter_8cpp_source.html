133void Filter::calcBPCoefficients()
134{

140 i64 phi_k = (M_PI*fCO_high*1000) / (SAMPLERATE_HZ/2);
141
142 size_t n = 0;
143 i64 mm_k = 0;
144 for(n = 0; n < NTAPS; n++)
145 {
146 mm_k = n*1000 - ((NTAPS-1)*1000)/2;
147 if(mm_k == 0)
148 {
149 taps_k_[n] = (phi_k-lambda_k)/M_PI;
150 }
151 else
152 {
153 taps_k_[n] = (1000000*(sin((mm_k*phi_k)/1000000.0)

156 }
157 }
