40 cout << "Save waveform data as \"" << path << "\"." << std::endl;
41
42 // Open output file
43 if(tinywav_open_write(
44 &tw,
45 1,
46 44100,
47 TW_FLOAT32,
48 TW_INLINE,
49 path.c_str()) != 0)
50 {
51 return false;

53
54 // Crop silence at end
55 vector<float> buf = audio_;
56 auto result = find_if(rbegin(buf), rend(buf),
57 [](float f)
58 {
59 return f!=0;
60 });
61 size_t idx_last = distance(result, rend(buf))-1;
62 size_t size = idx_last+1;
63
64 // Write
65 for (size_t i = 0; i < size; i++)
66 {
67 tinywav_write_f(&tw, static_cast<void*>(&(buf[i])), 1);
68 }
69
70 // Close file
71 tinywav_close_write(&tw);
72 return true;
