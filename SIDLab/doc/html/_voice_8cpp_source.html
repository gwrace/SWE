185void Voice::onSample(u64 a_sampleCtr)
186{
187 for(pair<const WaveformType, Waveform*>& wf : waveforms_)
188 {
189 wf.second->onSample();
190 }
191
192 switch(adsr_state_)
193 {
194 case ADSR::OUTSIDE:
195 if(gate_ && !gate_prev_) // GATE hit from complete silence
196 {
197 attack_.nSamples_ = attackNSamples_[attack_.idx_];
198 decay_.nSamples_ = decayReleaseNSamples_[decay_.idx_];
199 release_.nSamples_ = decayReleaseNSamples_[release_.idx_];
200 sustain_.level_ = sustain_.idx_;
201
202 adsr_state_ = ADSR::ATTACK;
203 attack_.begin_nSamples_ = a_sampleCtr + 1;
204 attack_.end_nSamples_ = a_sampleCtr + 1 + attack_.nSamples_;
205 attack_.m_s_ = scale_
206 / (attack_.end_nSamples_-attack_.begin_nSamples_);
207 attack_.offs_nSamples_ = 0;
208 break;
209 }
210 break;
211
212 case ADSR::ATTACK:
213 {
214 u64 nSamples_rel = a_sampleCtr - attack_.begin_nSamples_
215 + attack_.offs_nSamples_;
216 adsr_ampl_s_ = max((i64)0, (i64)attack_.m_s_*(i64)nSamples_rel);
217 attack_.peakAmpl_s_ = adsr_ampl_s_;
218
219 // Attack phase end, now decay
220 if(a_sampleCtr >= attack_.end_nSamples_ && adsr_ampl_s_ >= scale_)
221 {
222 sustain_.ampl_s_ = (sustain_.level_*attack_.peakAmpl_s_)/15;
223
224 adsr_state_ = ADSR::DECAY;
225 decay_.begin_nSamples_ = a_sampleCtr + 1;
226 decay_.end_nSamples_ = a_sampleCtr + 1 + decay_.nSamples_;
227 i64 delta_ampl_d_s = sustain_.ampl_s_ - (i64)attack_.peakAmpl_s_;
228 i64 delta_nSamples_d_ = decay_.end_nSamples_
229 - decay_.begin_nSamples_;
230 decay_.m_s_ = delta_ampl_d_s / delta_nSamples_d_;
231 decay_.b_s_ = attack_.peakAmpl_s_;
232 break;
233 }
234 // GATE released
235 if(gate_prev_ && !gate_)
236 {
237 adsr_state_ = ADSR::RELEASE;
238 release_.begin_nSamples_ = a_sampleCtr + 1;
239 release_.end_nSamples_ = a_sampleCtr + 1 + release_.nSamples_;
240 i64 delta_nSamples_r_ = release_.end_nSamples_
241 - release_.begin_nSamples_;
242 i64 delta_ampl_r_s = attack_.peakAmpl_s_;
243 release_.m_s_ = -delta_ampl_r_s / delta_nSamples_r_;
244 release_.b_s_ = attack_.peakAmpl_s_;
245 break;
246 }
247 }
248 break;
249
250 case ADSR::DECAY:
251 {
252 u64 nSamples_rel = a_sampleCtr - decay_.begin_nSamples_;
253 adsr_ampl_s_ = max((i64)0,
254 (i64)decay_.m_s_*(i64)nSamples_rel
255 + (i64)decay_.b_s_);
256 decay_.lastAmpl_s_ = adsr_ampl_s_;
257
258 // Decay phase end, now sustain (hold)
259 if(a_sampleCtr >= decay_.end_nSamples_
260 && adsr_ampl_s_ <= sustain_.ampl_s_)
261 {
262 adsr_state_ = ADSR::SUSTAIN;
263 break;
264 }
265 // GATE released
266 if(gate_prev_ && !gate_)
267 {
268 adsr_state_ = ADSR::RELEASE;
269 release_.begin_nSamples_ = a_sampleCtr + 1;
270 release_.end_nSamples_ = a_sampleCtr + 1 + release_.nSamples_;
271 i64 delta_nSamples_r_ = release_.end_nSamples_
272 - release_.begin_nSamples_;
273 i64 delta_ampl_r_s = decay_.lastAmpl_s_;
274 release_.m_s_ = -delta_ampl_r_s / delta_nSamples_r_;
275 release_.b_s_ = decay_.lastAmpl_s_;
276 break;
277 }
278 }
279 break;
280
281 case ADSR::SUSTAIN:
282 {
283 // GATE released, sustain phase end
284 if(gate_prev_ && !gate_)
285 {
286 adsr_state_ = ADSR::RELEASE;
287 release_.begin_nSamples_ = a_sampleCtr + 1;
288 release_.end_nSamples_ = a_sampleCtr + 1 + release_.nSamples_;
289 i64 delta_nSamples_r_ = release_.end_nSamples_
290 -release_.begin_nSamples_;
291 i64 delta_ampl_r_s = decay_.lastAmpl_s_;
292 release_.m_s_ = -delta_ampl_r_s / delta_nSamples_r_;
293 release_.b_s_ = decay_.lastAmpl_s_;
294 break;
295 }
296 }
297 break;
298
299 case ADSR::RELEASE:
300 {
301 u64 nSamples_rel = a_sampleCtr - release_.begin_nSamples_;
302 adsr_ampl_s_ = max((i64)0,
303 (i64)release_.m_s_*(i64)nSamples_rel
304 + (i64)release_.b_s_);
305 release_.lastAmpl_s_ = adsr_ampl_s_;
306
307 // Done, silence
308 if(a_sampleCtr >= release_.end_nSamples_ && adsr_ampl_s_ == 0)
309 {
310 adsr_state_ = ADSR::OUTSIDE;
311 adsr_ampl_s_ = 0;
312 break;
313 }
314 // GATE hit from unfinished release phase
315 if(!gate_prev_ && gate_)
316 {
317 attack_.nSamples_ = attackNSamples_[attack_.idx_];
318 decay_.nSamples_ = decayReleaseNSamples_[decay_.idx_];
319 release_.nSamples_ = decayReleaseNSamples_[release_.idx_];
320 sustain_.level_ = sustain_.idx_;
321
322 adsr_state_ = ADSR::ATTACK;
323 attack_.begin_nSamples_ = a_sampleCtr + 1;
324 attack_.m_s_ = scale_ / attack_.nSamples_;
325 u64 delta_ampl_s = max((i64)0, scale_ - (i64)release_.lastAmpl_s_);
326 attack_.end_nSamples_ = attack_.begin_nSamples_
327 + (attack_.nSamples_*delta_ampl_s)
328 /scale_;
329 attack_.offs_nSamples_ = (attack_.nSamples_ * release_.lastAmpl_s_)
330 / scale_;
331 break;
332 }
333 }
334 break;
335
336 default:
337 break;
338 }
339
340 u64 sum = 0;
341 u16 nWfEnabled = 0;
342 for(pair<const WaveformType, bool>& wf : waveform_enabled_)
343 {
344 int ena = (wf.second == true);
345 nWfEnabled += ena;
346 sum += (waveforms_[wf.first]->value())*ena;
347 }
348
349 output_ = 0.0;
350 gate_prev_ = gate_;
351
352 if(nWfEnabled == 0)
353 {
354 return;
355 }
356
357 u16 rawValue = sum / nWfEnabled;
358 i32 r = (i32)(rawValue) - 32768;
359 i64 radsr_scale = r*(i64)adsr_ampl_s_;
360 output_ = radsr_scale / scale_;
361
362 return;
363}
