190 float SID::onCLK()
191 {
192  sampleCtr_++;
193 
194  if(!pins_[NRES-1])
195  {
196  onNRES();
197  return 0.0;
198  }
199 
200  if(!pins_[NCS-1])
201  {
202  Address addr =
203  (pins_[A[0]-1] << 0)
204  + (pins_[A[1]-1] << 1)
205  + (pins_[A[2]-1] << 2)
206  + (pins_[A[3]-1] << 3)
207  + (pins_[A[4]-1] << 4);
208 
209  b8 data =
210  (pins_[D[0]-1] << 0)
211  + (pins_[D[1]-1] << 1)
212  + (pins_[D[2]-1] << 2)
213  + (pins_[D[3]-1] << 3)
214  + (pins_[D[4]-1] << 4)
215  + (pins_[D[5]-1] << 5)
216  + (pins_[D[6]-1] << 6)
217  + (pins_[D[7]-1] << 7);
218 
219  #if defined (REAL_SID_ATTACHED)
220  realSID_.send({sampleCtr_, static_cast<u8>(0b00100000 | addr), data});
221  #endif
222 
223  switch(addr)
224  {
225  case FC_LO:
226  {
227  filterCutoffFreqLo_ = data & 0b00000111;
228  b16 filterCutoffFreq
229  = ((b16)filterCutoffFreqHi_ << 3) | filterCutoffFreqLo_;
230  double fco = (30.0 + filterCutoffFreq*5.8);
231  filterBanks_[V1].setCutoffFreq(fco);
232  filterBanks_[V2].setCutoffFreq(fco);
233  filterBanks_[V3].setCutoffFreq(fco);
234  break;
235  }
236 
237  case FC_HI:
238  {
239  filterCutoffFreqHi_ = data;
240  b16 filterCutoffFreq
241  = ((b16)filterCutoffFreqHi_ << 3) | filterCutoffFreqLo_;
242  double fco = (30.0 + filterCutoffFreq*5.8);
243  filterBanks_[V1].setCutoffFreq(fco);
244  filterBanks_[V2].setCutoffFreq(fco);
245  filterBanks_[V3].setCutoffFreq(fco);
246  break;
247  }
248 
249  case RES_FILT:
250  voicePassFilter_[V1] = data & (1 << 0);
251  voicePassFilter_[V2] = data & (1 << 1);
252  voicePassFilter_[V3] = data & (1 << 2);
253  break;
254 
255  case MODE_VOL:
256  volume_ = (float)(data & 0b00001111)/15.0;
257  filterEnabled_[LP] = data & (1 << 4);
258  filterEnabled_[BP] = data & (1 << 5);
259  filterEnabled_[HP] = data & (1 << 6);
260  break;
261 
262  default:
263  if(voiceSetFcts_.count(addr))
264  {
265  voiceSetFcts_.at(addr)(data);
266  }
267  break;
268  }
269  }
270 
271  // Filter voices and mix
272  float audioOut = 0.0;
273  for(pair<const VoiceIdx, Voice>& voice : voices_)
274  {
275  voice.second.onSample(sampleCtr_);
276  VoiceIdx v = voice.first;
277  audioOut += filterBanks_[v].sample(
278  voice.second.output(),
279  voicePassFilter_[v] && filterEnabled_[LP],
280  voicePassFilter_[v] && filterEnabled_[BP],
281  voicePassFilter_[v] && filterEnabled_[HP]);
282  }
283 
284  audioOut *= volume_ / (static_cast<int>(N_VOICES)*32767.0);
285 
286  return audioOut;
287 }
